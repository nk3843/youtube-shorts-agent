import json

def validate_quality(last_output):
    """
    Parses the Judge's JSON output to decide next steps.
    """
    try:
        # Clean the output in case the model adds ```json fences
        clean_output = last_output.replace("```json", "").replace("```", "").strip()
        data = json.loads(clean_output)
        
        # The logic: If score is high enough, we are done.
        if data["total_score"] >= 8:
            return "approved" # Breaks the loop
        
        return "refine" # Continues to Refiner Agent
        
    except json.JSONDecodeError:
        # If the LLM messes up the JSON, send it back to refine just in case
        print("Error parsing Judge output, retrying loop...")
        return "refine"
